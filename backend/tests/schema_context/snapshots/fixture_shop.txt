shop."Order Events"
  id bigint PK
  "order" bigint → shop.orders.id
  "Event Type" text
  occurred_at timestamp with time zone
shop.customers
  id uuid PK
  full_name text
  email text
  phone text nullable
  city text
  region_id integer → shop.regions.id
  segment text
  signup_at timestamp with time zone
shop.order_items
  order_id bigint → shop.orders.id
  line_no integer
  product_id integer → shop.products.id
  quantity integer
  unit_price numeric(12,2)
  PRIMARY KEY (order_id, line_no)

shop.page_views
  id bigint PK
  customer_id uuid nullable
  path text
  viewed_at timestamp with time zone
shop.products
  id integer PK
  sku text UNIQUE
  name text
  category text
  unit_price numeric(12,2)
  is_active boolean
shop.refunds
  id bigint PK
  order_id bigint → shop.orders.id
  amount numeric(12,2)
  reason text nullable
  refunded_at timestamp without time zone
shop.regions
  id integer PK
  name text
  zone text
